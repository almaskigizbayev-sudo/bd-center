/** Полный список полей таблицы с учётом общих. */
function schemaFields_(tableName) {
  var t = TABLES[tableName];
  if (!t) throw new Error('Unknown table: ' + tableName);
  if (t.system) return [['id', 'id', 'ID']].concat(t.fields);
  // Для общих полей подставляем описание статусов конкретной таблицы
  var common = COMMON_FIELDS.map(function (f) {
    if (f[0] === 'status' && t.statuses) return ['status', 'text', t.statuses.join(' / ')];
    return f;
  });
  return [common[0]].concat(t.fields, common.slice(1));
}

function schemaHeaders_(tableName) {
  return schemaFields_(tableName).map(function (f) { return f[0]; });
}

var ROOT_FOLDER_NAME = 'BD COMMAND CENTER';
var DB_FILE_NAME = 'BD COMMAND CENTER — DB';

var DRIVE_TREE = {
  '00_SYSTEM': [],
  '01_FOUNDRY': ['Companies', 'RFQ', 'Quotes', 'Drawings'],
  '02_AGRIBUSINESS': ['Companies', 'Contracts'],
  '03_WAREHOUSING_LOGISTICS': ['Companies', 'Facilities', 'Leases'],

function setup() {
  var props = PropertiesService.getScriptProperties();
  var report = [];

  // 1. Владелец
  var me = Session.getEffectiveUser().getEmail();
  if (!props.getProperty('OWNER_EMAIL')) {
    props.setProperty('OWNER_EMAIL', me);
    report.push('OWNER_EMAIL = ' + me);
  }

  // 2. Папки Drive
  var root = getOrCreateRootFolder_(props, report);
  var folderIds = {};
  Object.keys(DRIVE_TREE).forEach(function (name) {
    var f = getOrCreateChild_(root, name, report);
    folderIds[name] = f.getId();
    DRIVE_TREE[name].forEach(function (sub) { getOrCreateChild_(f, sub, report); });
  });
  props.setProperty('FOLDER_IDS', JSON.stringify(folderIds));

  // 3. Мастер-таблица
  var ss = getOrCreateDb_(props, DriveApp.getFolderById(folderIds['00_SYSTEM']), report);
  ss.setSpreadsheetTimeZone('Asia/Almaty');

  // 4. Листы и заголовки
  Object.keys(TABLES).forEach(function (name) {
    ensureSheet_(ss, name, report);
  });
  removeDefaultSheet_(ss);

  // 5. Справочники и настройки (только если пусто)
  seedDirections_(ss, folderIds, report);
  seedIfEmpty_(ss, 'Settings', SEED_SETTINGS.map(function (s) {
    return { id: newId_('SET'), key: s[0], value: s[1], value_type: s[2], description: s[3] };
  }), report);
  seedIfEmpty_(ss, 'ReferenceData', SEED_REFERENCE.map(function (r) {
    var o = JSON.parse(JSON.stringify(r)); o.id = newId_('REF'); return o;
  }), report);

  // 6. Журнал
  appendRow_(ss, 'ActivityLog', {
    id: newId_('ACT'), created_at: new Date(), updated_at: new Date(), status: 'ok',
    owner: me, source: 'system', rev: 1, actor: 'system', action: 'setup',
    entity_type: 'System', note: report.length ? report.join('\n').slice(0, 45000) : 'Без изменений'
  });

  var msg = 'Готово.\nТаблица: ' + ss.getUrl() + '\nПапка: ' + root.getUrl() +
            '\n\nИзменения:\n' + (report.length ? report.join('\n') : 'ничего, всё уже было на месте');
  Logger.log(msg);
  return msg;
}

/* ---------- Drive ---------- */

function getOrCreateRootFolder_(props, report) {
  var id = props.getProperty('ROOT_FOLDER_ID');
  if (id) {
    try { var f = DriveApp.getFolderById(id); if (!f.isTrashed()) return f; } catch (e) { /* пересоздадим */ }
  }
  var it = DriveApp.getRootFolder().getFoldersByName(ROOT_FOLDER_NAME);
  var folder;
  if (it.hasNext()) {
    folder = it.next();
  } else {
    folder = DriveApp.getRootFolder().createFolder(ROOT_FOLDER_NAME);
    report.push('Папка: ' + ROOT_FOLDER_NAME);
  }
  props.setProperty('ROOT_FOLDER_ID', folder.getId());
  return folder;
}

function getOrCreateChild_(parent, name, report) {
  var it = parent.getFoldersByName(name);
  if (it.hasNext()) return it.next();
  report.push('Папка: ' + parent.getName() + '/' + name);
  return parent.createFolder(name);
}

/* ---------- Spreadsheet ---------- */

function getOrCreateDb_(props, systemFolder, report) {
  var id = props.getProperty('SPREADSHEET_ID');
  if (id) {
    try { return SpreadsheetApp.openById(id); } catch (e) { /* пересоздадим */ }
  }
  var it = systemFolder.getFilesByName(DB_FILE_NAME);
  var ss;
  if (it.hasNext()) {
    ss = SpreadsheetApp.open(it.next());
  } else {
    ss = SpreadsheetApp.create(DB_FILE_NAME);
    DriveApp.getFileById(ss.getId()).moveTo(systemFolder);
    report.push('Таблица: ' + DB_FILE_NAME);
  }
  props.setProperty('SPREADSHEET_ID', ss.getId());
  return ss;
}

function ensureSheet_(ss, name, report) {
  var fields = schemaFields_(name);
  var headers = fields.map(function (f) { return f[0]; });
  var sh = ss.getSheetByName(name);

  if (!sh) {
    sh = ss.insertSheet(name);
    sh.getRange(1, 1, 1, headers.length).setValues([headers]);
    sh.getRange(1, 1, 1, headers.length).setNotes([fields.map(function (f) { return f[2] + '\nТип: ' + f[1]; })]);
    styleSheet_(sh, fields, TABLES[name].group);
    report.push('Лист: ' + name + ' (' + headers.length + ' колонок)');
    return sh;
  }

  // Лист есть: добавляем недостающие колонки в конец, ничего не удаляем
  var lastCol = Math.max(sh.getLastColumn(), 1);
  var existing = sh.getRange(1, 1, 1, lastCol).getValues()[0].map(String);
  var missing = fields.filter(function (f) { return existing.indexOf(f[0]) === -1; });
  if (missing.length) {
    var start = existing.filter(String).length + 1;
    sh.getRange(1, start, 1, missing.length).setValues([missing.map(function (f) { return f[0]; })]);
    sh.getRange(1, start, 1, missing.length).setNotes([missing.map(function (f) { return f[2] + '\nТип: ' + f[1]; })]);
    report.push('Лист ' + name + ': добавлены колонки ' + missing.map(function (f) { return f[0]; }).join(', '));
  }
  return sh;
}

var GROUP_COLORS = { common: '#2B2F36', system: '#5A5F66', FDY: '#3A3F47', AGR: '#4B6B3C', WHL: '#4A5A78', WLF: '#7A4B3A' };

function styleSheet_(sh, fields, group) {
  var n = fields.length;
  sh.setFrozenRows(1);
  sh.setFrozenColumns(1);
  sh.getRange(1, 1, 1, n)
    .setFontWeight('bold').setFontColor('#FFFFFF')
    .setBackground(GROUP_COLORS[group] || '#2B2F36')
    .setWrap(false);
  sh.setTabColor(GROUP_COLORS[group] || '#2B2F36');

  var maxRows = sh.getMaxRows();
  fields.forEach(function (f, i) {
    var col = i + 1, type = f[1];
    var rng = sh.getRange(2, col, Math.max(maxRows - 1, 1), 1);
    if (type === 'datetime') rng.setNumberFormat('yyyy-mm-dd hh:mm');
    else if (type === 'date') rng.setNumberFormat('yyyy-mm-dd');
    else if (type === 'money') rng.setNumberFormat('#,##0.00');
    else if (type === 'percent') rng.setNumberFormat('0"%"');
    else if (type === 'id' || type === 'text' || type.indexOf('ref') === 0 || type.indexOf('enum') === 0 || type.indexOf('multi') === 0 || type === 'phone') rng.setNumberFormat('@');
  });

  // Предупреждение при ручной правке заголовков
  var p = sh.getRange(1, 1, 1, n).protect().setDescription('Заголовки схемы. Переименование сломает систему.');
  p.setWarningOnly(true);
}

function removeDefaultSheet_(ss) {
  ['Sheet1', 'Лист1', 'Лист 1'].forEach(function (nm) {
    var sh = ss.getSheetByName(nm);
    if (sh && ss.getSheets().length > 1 && sh.getLastRow() === 0) ss.deleteSheet(sh);
  });
}

/* ---------- Seed ---------- */

function seedDirections_(ss, folderIds, report) {
  var sh = ss.getSheetByName('BusinessDirections');
  if (sh.getLastRow() > 1) return;
  var rows = DIRECTIONS.map(function (d) {
    return { id: 'DIR-' + d.code, code: d.code, name: d.name, name_en: d.name_en, table_prefix: d.prefix,
             color: d.color, sort_order: d.order, drive_folder_id: folderIds[d.folder] || '', active: true };
  });
  writeObjects_(sh, rows);
  report.push('Направления: 4');
}

function seedIfEmpty_(ss, sheetName, objects, report) {
  var sh = ss.getSheetByName(sheetName);
  if (sh.getLastRow() > 1) return;
  writeObjects_(sh, objects);
  report.push(sheetName + ': ' + objects.length + ' строк');
}

function writeObjects_(sh, objects) {
  if (!objects.length) return;
  var headers = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0];
  var values = objects.map(function (o) {
    return headers.map(function (h) { return o[h] === undefined ? '' : o[h]; });
  });
  sh.getRange(sh.getLastRow() + 1, 1, values.length, headers.length).setValues(values);
}

function appendRow_(ss, sheetName, obj) {
  writeObjects_(ss.getSheetByName(sheetName), [obj]);
}

}

function ss_() {
  if (!__ss) {
    var id = prop_('SPREADSHEET_ID');
    if (!id) throw apiError_('NOT_SET_UP', 'База не развёрнута. Запустите setup() в редакторе Apps Script.');
    __ss = SpreadsheetApp.openById(id);
  }
  return __ss;
}

/** Настройки из листа Settings с приведением типов. */
function settings_() {
  if (__settings) return __settings;
  var sh = ss_().getSheetByName('Settings');
  var values = sh.getDataRange().getValues();
  var h = values[0];
  var iKey = h.indexOf('key'), iVal = h.indexOf('value'), iType = h.indexOf('value_type');
  var out = {};
  for (var r = 1; r < values.length; r++) {
    var k = String(values[r][iKey] || '').trim();
    if (!k) continue;
    var v = values[r][iVal], t = values[r][iType];
    if (t === 'number') v = Number(v);
    else if (t === 'bool') v = (v === true || String(v).toLowerCase() === 'true');
    else if (t === 'json') { try { v = JSON.parse(v); } catch (e) { /* оставляем строкой */ } }
    else if (v instanceof Date) v = Utilities.formatDate(v, 'Asia/Almaty', 'HH:mm');
    else v = String(v);
    out[k] = v;
  }
  __settings = out;
  return out;
}

function setting_(key, fallback) {
  var s = settings_();
  return s.hasOwnProperty(key) ? s[key] : fallback;
}

function ownerEmail_() {
  var e = prop_('OWNER_EMAIL');
  if (!e) throw apiError_('NOT_SET_UP', 'Не задан OWNER_EMAIL. Запустите setup().');
  return e;
}

  return e;
}

function sha256Hex_(text) {
  var bytes = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, String(text), Utilities.Charset.UTF_8);
  return bytes.map(function (b) { var v = (b < 0 ? b + 256 : b).toString(16); return v.length === 1 ? '0' + v : v; }).join('');
}

var LABEL_FIELDS = ['name', 'title', 'full_name', 'quote_number', 'item_name', 'description', 'body', 'period'];

/* ======================= Чтение ======================= */

function isWorkTable_(name) {
  return !!TABLES[name] && TABLES[name].group !== 'system';
}

function assertWorkTable_(name) {
  if (!isWorkTable_(name)) throw apiError_('BAD_TABLE', 'Таблица недоступна: ' + name);
}

function loadTable_(name, fresh) {
  if (!fresh && __tables[name]) return __tables[name];
  var sh = ss_().getSheetByName(name);
  if (!sh) throw apiError_('NOT_SET_UP', 'Нет листа ' + name + '. Запустите setup().');
  var values = sh.getDataRange().getValues();
  var headers = values[0].map(String);
  var rows = [], index = {};
  var iId = headers.indexOf('id');
  for (var r = 1; r < values.length; r++) {
    var id = String(values[r][iId] || '');
    if (!id) continue;
    var obj = {};
    for (var c = 0; c < headers.length; c++) if (headers[c]) obj[headers[c]] = values[r][c];
    obj._row = r + 1;
    index[id] = rows.length;
    rows.push(obj);
  }
  __tables[name] = { name: name, sheet: sh, headers: headers, rows: rows, index: index };
  return __tables[name];
}

function invalidate_(name) { delete __tables[name]; }

function findById_(name, id) {
  var t = loadTable_(name);
  var i = t.index[String(id)];
  return i === undefined ? null : t.rows[i];
}

/** Объект для отдачи в API: даты в ISO, служебные поля убраны. */
function serialize_(obj) {
  var out = {};
  Object.keys(obj).forEach(function (k) {
    if (k.charAt(0) === '_') return;
    var v = obj[k];
    out[k] = (v instanceof Date) ? (isNaN(v.getTime()) ? '' : v.toISOString()) : v;
  });
  return out;
}

function labelOf_(obj) {
  if (!obj) return '';
  for (var i = 0; i < LABEL_FIELDS.length; i++) {
    var v = obj[LABEL_FIELDS[i]];
    if (v !== '' && v !== null && v !== undefined) return String(v).slice(0, 80);
  }
  return String(obj.id);
}

/** Подписи для ссылочных полей: {field: {id: label}} */
function refLabels_(name, items) {
  var out = {};
  schemaFields_(name).forEach(function (f) {
    var m = String(f[1]).match(/^refs?:(.+)$/);
    if (!m || !TABLES[m[1]]) return;
    var ids = {};
    items.forEach(function (it) {
      String(it[f[0]] || '').split(',').forEach(function (x) { x = x.trim(); if (x) ids[x] = true; });
    });
    var keys = Object.keys(ids);
    if (!keys.length) return;
    var map = {};
    keys.forEach(function (id) { var o = findById_(m[1], id); map[id] = o ? labelOf_(o) : '(не найдено)'; });
    out[f[0]] = map;
  });
  return out;
}

/* ======================= Список и поиск ======================= */

function dbList_(name, opts) {
  assertWorkTable_(name);
  opts = opts || {};
  var t = loadTable_(name);
  var q = String(opts.q || '').toLowerCase().trim();
  var filters = opts.filters || {};
  var items = t.rows.filter(function (r) {
    if (!opts.includeArchived && (r.status === 'archived' || r.deleted_at)) return false;
    if (opts.direction && String(r.direction || '').split(',').map(function (s) { return s.trim(); }).indexOf(opts.direction) === -1) return false;
    if (opts.status && r.status !== opts.status) return false;
    for (var k in filters) {
      if (!filters.hasOwnProperty(k) || filters[k] === '' || filters[k] === null) continue;
      if (String(r[k]).split(',').map(function (s) { return s.trim(); }).indexOf(String(filters[k])) === -1) return false;
    }
    if (q) return rowText_(r).indexOf(q) !== -1;
    return true;
  });

  var sort = opts.sort || '-updated_at';
  var desc = sort.charAt(0) === '-', key = desc ? sort.slice(1) : sort;
  items.sort(function (a, b) {
    var x = a[key], y = b[key];
    if (x instanceof Date) x = x.getTime();
    if (y instanceof Date) y = y.getTime();
    if (x === '' || x === undefined) return 1;
    if (y === '' || y === undefined) return -1;
    return (x < y ? -1 : x > y ? 1 : 0) * (desc ? -1 : 1);
  });

  var total = items.length;
  var offset = Math.max(0, Number(opts.offset) || 0);
  var limit = Math.min(500, Math.max(1, Number(opts.limit) || 200));
  var page = items.slice(offset, offset + limit).map(serialize_);
  return { items: page, total: total, labels: refLabels_(name, page) };
}

function rowText_(r) {
  var s = [];
  for (var k in r) if (r.hasOwnProperty(k) && k.charAt(0) !== '_' && typeof r[k] === 'string') s.push(r[k]);
  return s.join(' \u0001 ').toLowerCase();
}

var SEARCH_TABLES = ['Companies', 'Contacts', 'Projects', 'Opportunities', 'Tasks', 'Meetings', 'Agreements', 'Documents', 'Notes'];

function dbSearch_(q) {
  q = String(q || '').trim();
  if (q.length < 2) return { results: [] };
  var results = [];
  SEARCH_TABLES.forEach(function (name) {
    var res = dbList_(name, { q: q, limit: 8 });
    res.items.forEach(function (it) { results.push({ table: name, id: it.id, label: labelOf_(it), status: it.status, direction: it.direction }); });
  });
  return { results: results };
}

/* ======================= Карточка ======================= */

function dbGet_(name, id) {
  assertWorkTable_(name);
  var rec = findById_(name, id);
  if (!rec) throw apiError_('NOT_FOUND', 'Запись не найдена: ' + id);
  var out = { record: serialize_(rec), labels: refLabels_(name, [rec]), related: [], activity: [] };

  // Все таблицы, у которых есть ссылка на эту
  Object.keys(TABLES).forEach(function (other) {
    if (!isWorkTable_(other)) return;
    var refFields = schemaFields_(other).filter(function (f) {
      var m = String(f[1]).match(/^refs?:(.+)$/);
      return m && m[1] === name;
    }).map(function (f) { return f[0]; });
    var hasGeneric = schemaHeaders_(other).indexOf('related_id') !== -1;
    if (!refFields.length && !hasGeneric) return;

    var t = loadTable_(other);
    var hits = t.rows.filter(function (r) {
      if (r.status === 'archived' || r.deleted_at) return false;
      if (hasGeneric && String(r.related_id) === String(id)) return true;
      return refFields.some(function (f) {
        return String(r[f] || '').split(',').map(function (s) { return s.trim(); }).indexOf(String(id)) !== -1;
      });
    });
    if (!hits.length) return;
    hits.sort(function (a, b) { return (b.updated_at || 0) - (a.updated_at || 0); });
    out.related.push({
      table: other, total: hits.length,
      items: hits.slice(0, 25).map(function (r) { return { id: r.id, label: labelOf_(r), status: r.status, updated_at: serialize_(r).updated_at }; })
    });
  });

  var log = loadTable_('ActivityLog').rows.filter(function (r) { return String(r.entity_id) === String(id); });
  out.activity = log.slice(-30).reverse().map(function (r) {
    return { at: serialize_(r).created_at, actor: r.actor, action: r.action, diff: r.diff_json, note: r.note };
  });
  return out;
}

/* ======================= Запись ======================= */

function withLock_(fn) {
  var lock = LockService.getScriptLock();
  if (!lock.tryLock(10000)) throw apiError_('BUSY', 'База занята другим запросом, повторите через пару секунд.');
  try { return fn(); } finally { lock.releaseLock(); }
}

function dbCreate_(name, input, opts) {
  assertWorkTable_(name);
  opts = opts || {};
  var def = TABLES[name];
  var rec = sanitize_(name, input || {}, false);
  var now = new Date();

  rec.id = newId_(def.prefix);
  rec.created_at = now;
  rec.updated_at = now;
  rec.rev = 1;
  rec.status = rec.status || def.statuses[0];
  rec.owner = rec.owner || ownerEmail_();
  rec.source = opts.source || 'manual';
  if (name === 'Companies') rec.name_norm = normName_(rec.name);

  (REQUIRED_FIELDS[name] || []).forEach(function (f) {
    if (rec[f] === '' || rec[f] === undefined || rec[f] === null) {
      throw apiError_('VALIDATION', 'Не заполнено обязательное поле', { field: f });
    }
  });

  return withLock_(function () {
    invalidate_(name);
    var t = loadTable_(name, true);
    if (!opts.force) {
      var dup = findDuplicate_(name, rec, t);
      if (dup) throw apiError_('DUPLICATE', 'Похоже, такая запись уже есть', { existing: serialize_(dup), label: labelOf_(dup) });
    }
    appendObjects_(t, [rec]);
    invalidate_(name);
    logActivity_(opts.actor || 'owner', 'create', name, rec.id, null, opts.note || '');
    return { record: serialize_(rec) };
  });
}

function dbUpdate_(name, id, rev, patch, opts) {
  assertWorkTable_(name);
  opts = opts || {};
  var clean = sanitize_(name, patch || {}, true);
  if (!Object.keys(clean).length) throw apiError_('VALIDATION', 'Нет изменений');

  return withLock_(function () {
    invalidate_(name);
    var t = loadTable_(name, true);
    var cur = findById_(name, id);
    if (!cur) throw apiError_('NOT_FOUND', 'Запись не найдена: ' + id);
    if (rev !== undefined && rev !== null && Number(cur.rev) !== Number(rev)) {
      throw apiError_('CONFLICT', 'Запись изменилась с момента открытия. Обновите карточку.', { current: serialize_(cur) });
    }
    var diff = {};
    Object.keys(clean).forEach(function (k) {
      var before = cur[k], after = clean[k];
      var b = before instanceof Date ? before.getTime() : before;
      var a = after instanceof Date ? after.getTime() : after;
      if (String(b) !== String(a)) {
        diff[k] = [serialize_({ v: before }).v, serialize_({ v: after }).v];
        cur[k] = after;
      }
    });
    if (!Object.keys(diff).length) return { record: serialize_(cur), unchanged: true };

    (REQUIRED_FIELDS[name] || []).forEach(function (f) {
      if (cur[f] === '' || cur[f] === null || cur[f] === undefined) throw apiError_('VALIDATION', 'Нельзя очистить обязательное поле', { field: f });
    });
    if (name === 'Companies' && diff.name) cur.name_norm = normName_(cur.name);
    if (opts.archive) { cur.deleted_at = new Date(); }
    if (opts.restore) { cur.deleted_at = ''; }
    cur.updated_at = new Date();
    cur.rev = Number(cur.rev || 0) + 1;

    writeRow_(t, cur);
    invalidate_(name);
    logActivity_(opts.actor || 'owner', opts.action || 'update', name, id, diff, '');
    return { record: serialize_(cur) };
  });
}

function dbArchive_(name, id, rev) {
  return dbUpdate_(name, id, rev, { status: 'archived' }, { archive: true, action: 'archive' });
}

function dbRestore_(name, id, rev) {
  return dbUpdate_(name, id, rev, { status: TABLES[name].statuses[0] }, { restore: true, action: 'restore' });
}

function appendObjects_(t, objs) {
  var values = objs.map(function (o) { return t.headers.map(function (h) { return cellValue_(o[h]); }); });
  var start = t.sheet.getLastRow() + 1;
  t.sheet.getRange(start, 1, values.length, t.headers.length).setValues(values);
}

function writeRow_(t, obj) {
  var row = t.headers.map(function (h) { return cellValue_(obj[h]); });
  t.sheet.getRange(obj._row, 1, 1, t.headers.length).setValues([row]);
}

/** Строки, начинающиеся с = + - @, Sheets может принять за формулу. Апостроф делает их текстом. */
function cellValue_(v) {
  if (v === undefined || v === null) return '';
  if (typeof v === 'string' && /^[=+\-@]/.test(v)) return "'" + v;
  return v;
}

/* ======================= Валидация ======================= */

function refCodes_() {
  if (__refCodes) return __refCodes;
  var sh = ss_().getSheetByName('ReferenceData');
  var values = sh.getDataRange().getValues();
  var h = values[0], iL = h.indexOf('list_name'), iC = h.indexOf('code'), iA = h.indexOf('active');
  var out = {};
  for (var r = 1; r < values.length; r++) {
    if (values[r][iA] === false || String(values[r][iA]).toLowerCase() === 'false') continue;
    var l = String(values[r][iL]);
    (out[l] = out[l] || {})[String(values[r][iC])] = true;
  }
  __refCodes = out;
  return out;
}

function sanitize_(name, input, partial) {
  var fields = schemaFields_(name), out = {};
  var def = TABLES[name];
  fields.forEach(function (f) {
    var key = f[0], type = String(f[1]);
    if (!input.hasOwnProperty(key)) return;
    if (SYSTEM_FIELDS.indexOf(key) !== -1) return;
    var v = input[key];
    if (v === null || v === undefined) v = '';
    if (typeof v === 'string') v = v.trim();
    if (v === '') { out[key] = ''; return; }

    if (key === 'status') {
      if (def.statuses.indexOf(v) === -1 && v !== 'archived') throw apiError_('VALIDATION', 'Недопустимый статус', { field: key, value: v });
      out[key] = v; return;
    }
    out[key] = coerce_(key, type, v);
  });
  return out;
}

function coerce_(key, type, v) {
  var fail = function (msg) { throw apiError_('VALIDATION', msg, { field: key, value: v }); };
  if (type === 'number' || type === 'money' || type === 'percent') {
    var n = typeof v === 'number' ? v : Number(String(v).replace(/\s/g, '').replace(',', '.'));
    if (isNaN(n)) fail('Ожидается число');
    return n;
  }
  if (type === 'date' || type === 'datetime') {
    var d = v instanceof Date ? v : new Date(v);
    if (isNaN(d.getTime())) fail('Неверная дата');
    return d;
  }
  if (type === 'bool') return v === true || String(v).toLowerCase() === 'true';
  if (type === 'email') {
    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(String(v))) fail('Неверный email');
    return String(v).toLowerCase();
  }
  if (type === 'json') return typeof v === 'string' ? v : JSON.stringify(v);

  var m = type.match(/^(enum|multi):(.+)$/);
  if (m) {
    var allowed = refCodes_()[m[2]] || {};
    var parts = m[1] === 'multi' ? String(v).split(',').map(function (s) { return s.trim(); }).filter(String) : [String(v)];
    parts.forEach(function (p) { if (!allowed[p]) fail('Значения нет в справочнике ' + m[2]); });
    return parts.join(',');
  }
  m = type.match(/^(ref|refs):(.+)$/);
  if (m) {
    var ids = m[1] === 'refs' ? String(v).split(',').map(function (s) { return s.trim(); }).filter(String) : [String(v)];
    ids.forEach(function (id) { if (!findById_(m[2], id)) fail('Связанная запись не найдена в ' + m[2]); });
    return ids.join(',');
  }
  return String(v).slice(0, 49000);
}

/* ======================= Дубли ======================= */

function normName_(s) {
  return String(s || '').toLowerCase()
    .replace(/ё/g, 'е')
    .replace(/[«»"'“”„`]/g, ' ')
    .replace(/(^|\s)(тоо|ооо|ао|зао|оао|пао|ип|llp|llc|ltd|jsc|inc|gmbh)(?=\s|$|[.,])/g, ' ')
    .replace(/[.,()]/g, ' ')
    .replace(/\s+/g, ' ')
    .trim();
}

function digits_(s) { return String(s || '').replace(/\D/g, ''); }

function findDuplicate_(name, rec, t) {
  var live = t.rows.filter(function (r) { return r.status !== 'archived' && !r.deleted_at; });
  if (name === 'Companies') {
    return live.filter(function (r) {
      if (rec.tax_id && r.tax_id && String(r.tax_id).trim() === String(rec.tax_id).trim()) return true;
      return r.name_norm && r.name_norm === rec.name_norm && String(r.country || '') === String(rec.country || '');
    })[0] || null;
  }
  if (name === 'Contacts') {
    return live.filter(function (r) {
      if (rec.email && r.email && String(r.email).toLowerCase() === rec.email) return true;
      var p1 = digits_(rec.phone), p2 = digits_(r.phone);
      if (p1.length >= 10 && p2.length >= 10 && p1.slice(-10) === p2.slice(-10)) return true;
      return rec.full_name && String(r.full_name).toLowerCase().trim() === String(rec.full_name).toLowerCase() &&
             String(r.company_id || '') === String(rec.company_id || '');
    })[0] || null;
  }
  if (name === 'Documents' && rec.drive_file_id) {
    return live.filter(function (r) { return r.drive_file_id === rec.drive_file_id; })[0] || null;
  }
  return null;
}

/**
 * BD COMMAND CENTER — Auth.gs
 * Вход одноразовым кодом, который приходит ТОЛЬКО на OWNER_EMAIL.
 * В Script Properties хранится лишь SHA-256 хеш токена сессии.
 */

var SESSION_DAYS = 30;
var CODE_TTL_SEC = 600;
var CODE_MAX_TRIES = 5;
var CODES_PER_HOUR = 5;

function authRequestCode_() {
  var cache = CacheService.getScriptCache();
  var hourKey = 'auth:codes:' + Utilities.formatDate(new Date(), 'UTC', 'yyyyMMddHH');
  var sent = Number(cache.get(hourKey) || 0);
  if (sent >= CODES_PER_HOUR) throw apiError_('RATE_LIMIT', 'Слишком много запросов кода. Попробуйте через час.');

  var hex = Utilities.getUuid().replace(/-/g, '').slice(0, 10);
  var code = ('000000' + (parseInt(hex, 16) % 1000000)).slice(-6);
  cache.put('auth:code', JSON.stringify({ h: sha256Hex_(code), tries: 0, exp: Date.now() + CODE_TTL_SEC * 1000 }), CODE_TTL_SEC);
  cache.put(hourKey, String(sent + 1), 3600);

  var owner = ownerEmail_();
  MailApp.sendEmail({
    to: owner,
    subject: 'BD Command Center: код входа ' + code,
    body: 'Код входа: ' + code + '\n\nДействует 10 минут. Если вы не запрашивали код, просто проигнорируйте письмо: без кода войти нельзя.',
    htmlBody: '<div style="font-family:-apple-system,Segoe UI,Arial,sans-serif;color:#111">' +
      '<p style="margin:0 0 8px">Код входа в BD Command Center:</p>' +
      '<p style="font-size:32px;letter-spacing:6px;font-weight:600;margin:0 0 16px">' + code + '</p>' +
      '<p style="color:#666;margin:0">Действует 10 минут. Если вы не запрашивали код, проигнорируйте письмо.</p></div>'
  });
  logActivity_('system', 'login_code', 'Auth', '', null, 'Код отправлен');
  return { sentTo: maskEmail_(owner), ttlSec: CODE_TTL_SEC };
}

function authVerify_(code, device) {
  var cache = CacheService.getScriptCache();
  var raw = cache.get('auth:code');
  if (!raw) throw apiError_('CODE_EXPIRED', 'Код истёк или не запрашивался. Запросите новый.');
  var st = JSON.parse(raw);
  if (Date.now() > st.exp) { cache.remove('auth:code'); throw apiError_('CODE_EXPIRED', 'Код истёк. Запросите новый.'); }
  if (st.tries >= CODE_MAX_TRIES) { cache.remove('auth:code'); throw apiError_('CODE_LOCKED', 'Слишком много попыток. Запросите новый код.'); }

  if (sha256Hex_(String(code || '').replace(/\D/g, '')) !== st.h) {
    st.tries++;
    var left = Math.max(1, Math.round((st.exp - Date.now()) / 1000));
    cache.put('auth:code', JSON.stringify(st), left);
    throw apiError_('CODE_INVALID', 'Неверный код. Осталось попыток: ' + (CODE_MAX_TRIES - st.tries));
  }
  cache.remove('auth:code');

  var token = (Utilities.getUuid() + Utilities.getUuid()).replace(/-/g, '');
  var now = Date.now();
  var session = { exp: now + SESSION_DAYS * 86400000, created: now, device: String(device || 'Устройство').slice(0, 60) };
  PropertiesService.getScriptProperties().setProperty('SESS_' + sha256Hex_(token), JSON.stringify(session));
  logActivity_('owner', 'login', 'Auth', '', null, 'Вход: ' + session.device);
  return { token: token, expiresAt: new Date(session.exp).toISOString(), owner: maskEmail_(ownerEmail_()) };
}

function authCheck_(token) {
  if (!token) throw apiError_('AUTH_REQUIRED', 'Нужен вход');
  var props = PropertiesService.getScriptProperties();
  var key = 'SESS_' + sha256Hex_(token);
  var raw = props.getProperty(key);
  if (!raw) throw apiError_('AUTH_REQUIRED', 'Сессия не найдена. Войдите заново.');
  var s = JSON.parse(raw);
  if (Date.now() > s.exp) { props.deleteProperty(key); throw apiError_('AUTH_REQUIRED', 'Сессия истекла. Войдите заново.'); }
  s.key = key;
  return s;
}

function authLogout_(session) {
  PropertiesService.getScriptProperties().deleteProperty(session.key);
  logActivity_('owner', 'logout', 'Auth', '', null, session.device);
  return { done: true };
}

function authLogoutAll_() {
  var props = PropertiesService.getScriptProperties();
  var keys = props.getKeys().filter(function (k) { return k.indexOf('SESS_') === 0; });
  keys.forEach(function (k) { props.deleteProperty(k); });
  logActivity_('owner', 'logout_all', 'Auth', '', null, 'Сессий закрыто: ' + keys.length);
  return { closed: keys.length };
}

function authSessions_(current) {
  var props = PropertiesService.getScriptProperties().getProperties();
  return {

}

function maskEmail_(e) {
  var p = String(e).split('@');

}

/* ======================== Api ======================== */
/**
 * BD COMMAND CENTER — Api.gs
 * Единая точка входа для приложения. Запрос: POST, тело JSON
 *   { action: 'records.list', token: '...', payload: {...} }
 * Ответ: { ok: true, data } или { ok: false, error: { code, message, details } }
 */

function doGet() {
  return json_({ ok: true, data: { service: 'BD COMMAND CENTER API', version: API_VERSION } });
}

function doPost(e) {
  var req;
  try {
    req = JSON.parse(e && e.postData && e.postData.contents || '{}');
  } catch (err) {
    return json_({ ok: false, error: { code: 'BAD_REQUEST', message: 'Неверный JSON' } });
  }
  return json_(handle_(req));
}

function json_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}

function handle_(req) {
  var t0 = Date.now();
  var action = String(req.action || '');
  var p = req.payload || {};
  try {
    var data;
    if (action === 'auth.requestCode') data = authRequestCode_();
    else if (action === 'auth.verify') data = authVerify_(p.code, p.device);
    else {
      var session = authCheck_(req.token);
      switch (action) {
        case 'meta.get':        data = metaGet_(); break;
        case 'records.list':    data = dbList_(p.table, p); break;
        case 'records.get':     data = dbGet_(p.table, p.id); break;
        case 'records.create':  data = dbCreate_(p.table, p.data, { force: !!p.force }); break;
        case 'records.update':  data = dbUpdate_(p.table, p.id, p.rev, p.data); break;
        case 'records.archive': data = dbArchive_(p.table, p.id, p.rev); break;
        case 'records.restore': data = dbRestore_(p.table, p.id, p.rev); break;
        case 'search.global':   data = dbSearch_(p.q); break;
        case 'auth.sessions':   data = authSessions_(session); break;
        case 'auth.logout':     data = authLogout_(session); break;
        case 'auth.logoutAll':  data = authLogoutAll_(); break;
        default: throw apiError_('UNKNOWN_ACTION', 'Неизвестное действие: ' + action);
      }
    }
    return { ok: true, data: data, ms: Date.now() - t0 };
  } catch (err) {
    if (err && err.isApiError) {
      return { ok: false, error: { code: err.code, message: err.message, details: err.details }, ms: Date.now() - t0 };
    }
    logError_('api:' + action, err);
    return { ok: false, error: { code: 'SERVER_ERROR', message: 'Внутренняя ошибка сервера. Подробности в журнале (ActivityLog).' }, ms: Date.now() - t0 };
  }
}

/** Всё, что приложению нужно знать о структуре: таблицы, поля, справочники, направления. */
function metaGet_() {
  var schema = {};
  Object.keys(TABLES).forEach(function (name) {
    var t = TABLES[name];
    if (t.group === 'system') return;
    schema[name] = {
      prefix: t.prefix, group: t.group, desc: t.desc, statuses: t.statuses,
      required: REQUIRED_FIELDS[name] || [],
      fields: schemaFields_(name).map(function (f) { return { name: f[0], type: f[1], desc: f[2] }; })
    };
  });

  var refSheet = ss_().getSheetByName('ReferenceData').getDataRange().getValues();
  var h = refSheet[0], reference = {};
  for (var r = 1; r < refSheet.length; r++) {
    var row = {};
    h.forEach(function (k, i) { row[k] = refSheet[r][i]; });
    if (row.active === false || String(row.active).toLowerCase() === 'false') continue;
    (reference[row.list_name] = reference[row.list_name] || []).push({
      code: String(row.code), label: String(row.label_ru || row.code), direction: String(row.direction || ''), order: Number(row.sort_order) || 0
    });
  }
  Object.keys(reference).forEach(function (k) { reference[k].sort(function (a, b) { return a.order - b.order; }); });

  return {
    apiVersion: API_VERSION,
    owner: maskEmail_(ownerEmail_()),
    directions: DIRECTIONS.map(function (d) { return { code: d.code, name: d.name, prefix: d.prefix, color: d.color }; }),
    schema: schema,
    reference: reference,
    settings: { timezone: setting_('timezone', 'Asia/Almaty'), currency: setting_('ui.default_currency', 'KZT'), theme: setting_('ui.theme', 'auto') }
  };
}
